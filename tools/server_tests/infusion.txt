# Arcane Infusion, at midnight on a stone floor: an altar set by command with a diamond sword on it
# and seven Fire Essence in its basin (its block data read back); broken, it drops the sword. An
# infused iron sword on the ground (its component and coloured name). A zombie holding a
# fire-infused sword strikes a still villager (Resistance IV keeps it alive): the villager burns
# (its Fire and health logged).
gamerule doMobSpawning false
gamerule doMobLoot true
gamerule doDaylightCycle false
time set midnight
forceload add -8 -8 8 8
fill -6 99 -6 6 99 6 minecraft:stone
fill -6 100 -6 6 104 6 minecraft:air
kill @e[type=!player]
setblock 0 100 0 elementalarcana:infusion_altar{item:{id:"minecraft:diamond_sword",count:1},element:"fire",charge:7}
data get block 0 100 0
setblock 0 100 0 minecraft:air destroy
sleep 1
execute as @e[type=item] run data get entity @s Item
kill @e[type=item]
summon minecraft:item 0 101 2 {Item:{id:"minecraft:iron_sword",count:1,components:{"elementalarcana:infusion":"ice"}}}
execute as @e[type=item] run data get entity @s Item.components
kill @e[type=item]
summon minecraft:villager 3 100 0 {NoAI:1b,Tags:["eatest"],active_effects:[{id:"minecraft:resistance",amplifier:3b,duration:2000}]}
summon minecraft:zombie 1 100 0 {HandItems:[{id:"minecraft:iron_sword",count:1,components:{"elementalarcana:infusion":"fire"}},{}],Tags:["eatest"]}
sleep 3
data get entity @e[type=minecraft:villager,tag=eatest,limit=1] Fire
data get entity @e[type=minecraft:villager,tag=eatest,limit=1] Health
kill @e[type=!player]
fill -6 99 -6 6 104 6 minecraft:air
forceload remove all
gamerule doDaylightCycle true
time set day
