# Relics (see the spec): each given and bound by use (a shot of the ring of light), then its power
# tried, in survival with numbers in the log:
# 1: Feather of the Gale: a jump, then a second in mid-air (heights through the jump), no fall damage
#    from 20 blocks up (health before and after);
# 2: Ember Heart: standing in fire (health before and after, still burning or not);
# 3: Stoneheart Idol: a hard hit (absorption after); 4: Rimeheart Locket: slowness can't be given;
# 5: Storm Sigil: movement speed; 6: Sunstone: night vision, and a zombie 3 blocks off smoulders;
# 7: Revenant's Phylactery: a killing blow survived at 1 health, the next one not (recharging).
hud on
cmd time set noon
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd gamerule doMobLoot false
cmd gamerule doFireTick false
cmd gamemode survival
cmd difficulty normal
cmd effect clear @s
cmd effect give @s minecraft:instant_health 1 10 true
cmd tp @s 0.5 100 0.5 -90 5
wait 40
cmd kill @e[type=!player]
cmd kill @e[type=item]
cmd fill -8 99 -8 8 99 8 minecraft:grass_block
cmd fill -8 100 -8 8 130 8 minecraft:air
cmd clear @s
cmd arcana tree reset
wait 10
# 1: The Feather.
cmd give @s elementalarcana:feather_of_the_gale
wait 5
slot 0
use
wait 10
shot relic_bound
cmd tp @s 0.5 100 0.5 -90 5
wait 10
cmd execute as @s run data get entity @s Pos[1]
key jump down
wait 2
key jump up
wait 5
cmd execute as @s run data get entity @s Pos[1]
key jump down
wait 2
key jump up
wait 6
cmd execute as @s run data get entity @s Pos[1]
shot relic_double_jump
wait 30
cmd execute as @s run data get entity @s Health
cmd tp @s 0.5 120 0.5 -90 5
wait 50
cmd execute as @s run data get entity @s Health
# 2: The Ember Heart.
cmd give @s elementalarcana:ember_heart
wait 5
slot 1
use
wait 10
cmd setblock 0 100 0 minecraft:fire
cmd tp @s 0.5 100 0.5 -90 5
wait 40
shot relic_ember_fire
cmd execute as @s run data get entity @s Health
cmd execute as @s run data get entity @s Fire
cmd setblock 0 100 0 minecraft:air
# 3: The Stoneheart Idol.
cmd give @s elementalarcana:stoneheart_idol
wait 5
slot 2
use
wait 10
cmd damage @s 6 minecraft:generic
wait 2
cmd execute as @s run data get entity @s AbsorptionAmount
cmd effect give @s minecraft:instant_health 1 10 true
# 4: The Rimeheart Locket.
cmd give @s elementalarcana:rimeheart_locket
wait 5
slot 3
use
wait 10
cmd effect give @s minecraft:slowness 10 1
wait 2
cmd execute as @s run data get entity @s active_effects
# 5: The Storm Sigil.
cmd give @s elementalarcana:storm_sigil
wait 5
slot 4
use
wait 30
cmd attribute @s minecraft:generic.movement_speed get
# 6: The Sunstone.
cmd give @s elementalarcana:sunstone
wait 5
slot 5
use
cmd summon zombie 3.5 100 0.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b,Health:20f}
wait 70
shot relic_sunstone
cmd execute as @e[tag=eat] run data get entity @s Health
cmd execute as @s run data get entity @s active_effects
cmd tp @e[tag=eat] 0.5 -100 0.5
cmd kill @e[tag=eat]
wait 10
# 7: The Phylactery (bound looking up, away from anything the crosshair could catch).
cmd give @s elementalarcana:revenants_phylactery
wait 5
slot 6
look -90 -60
wait 2
use
wait 10
cmd damage @s 1000 minecraft:generic
wait 4
shot relic_phylactery
cmd execute as @s run data get entity @s Health
wait 20
cmd damage @s 1000 minecraft:generic
wait 10
cmd execute as @s run data get entity @s Health
wait 40
cmd effect clear @s
cmd clear @s
quit
