# rolled eight times (its Ember Heart now and then) and a Revenant's loot four times (its Phylactery

summon item 7 101 0 {Item:{id:"elementalarcana:sunstone",count:1}}
summon item 8 101 0 {Item:{id:"elementalarcana:revenants_phylactery",count:1}}
