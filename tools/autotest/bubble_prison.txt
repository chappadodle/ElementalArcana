# Bubble Prison: a husk in front of the player is trapped (shot of the bubble), then a hit pops it
# (shot of the burst); a second bubble runs out by itself. Its sounds (the cast's whoosh, the bloop,
# the gurgle while it holds, the pop and splash) can't be heard here: the run only shows nothing
# errors. The husk's health goes to the log.
hud on
cmd time set noon
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd gamerule doMobLoot false
cmd gamemode survival
cmd effect clear @s
cmd effect give @s minecraft:instant_health 1 10 true
cmd tp @s 0.5 100 0.5 -90 0
wait 40
cmd kill @e[type=!player]
cmd kill @e[type=item]
cmd fill -8 99 -12 30 99 12 minecraft:grass_block
cmd fill -8 100 -12 30 110 12 minecraft:air
cmd arcana tree reset
cmd arcana affinity reset
cmd arcana awaken water
cmd arcana level set 20
cmd arcana spell elementalarcana:bubble_prison 1
cmd arcana mana fill
cmd arcana cooldowns reset
cmd title @s clear
cmd summon husk 5.5 100 0.5 {Tags:["eat"],PersistenceRequired:1b,Health:200f,attributes:[{id:"minecraft:generic.max_health",base:200d}]}
cmd effect give @e[type=husk] minecraft:slowness 120 255 true
cmd effect give @e[type=husk] minecraft:weakness 120 255 true
wait 10
cmd tp @s 0.5 100 0.5 -90 10
select elementalarcana:bubble_prison
press
release
wait 14
shot bubble_formed
wait 20
cmd damage @e[type=husk,limit=1] 1 minecraft:player_attack by @s
wait 2
shot bubble_popped
wait 20
cmd execute as @e[type=husk] run data get entity @s Health
cmd arcana cooldowns reset
cmd arcana mana fill
press
release
wait 100
shot bubble_ran_out
cmd kill @e[type=husk]
cmd gamerule doMobLoot true
wait 10
quit
