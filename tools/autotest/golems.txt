# Elemental Golems: each of the four summoned 5 blocks in front of the player, who strikes it (by
# command) to make it a foe; it walks up, raises its arms (the warning) and slams: shots every 4
# ticks through the fight, third person. A villager stands beside the player (golems hurt anyone
# but monsters): its health and the player's position after each slam go to the log (Water throws
# back, Wind throws up, Fire burns, Earth sends a Tremor).
hud on
cmd time set noon
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd gamerule doMobLoot false
cmd gamemode survival
cmd effect clear @s
cmd effect give @s minecraft:instant_health 1 10 true
cmd effect give @s minecraft:resistance 900 4 true
cmd effect give @s minecraft:fire_resistance 900 0 true
cmd tp @s 0.5 100 0.5 -90 10
wait 40
cmd kill @e[type=!player]
cmd kill @e[type=item]
cmd fill -10 99 -14 30 99 14 minecraft:grass_block
cmd fill -10 100 -14 30 110 14 minecraft:air
cmd arcana tree reset
cmd arcana affinity reset
cmd title @s clear
wait 20
# The fire golem.
cmd tp @s 0.5 100 0.5 -90 10
cmd arcana golem fire
cmd summon villager 0.5 100 2.5 {NoAI:1b,Tags:["eat"]}
wait 10
cmd damage @e[type=elementalarcana:fire_golem,limit=1] 1 minecraft:player_attack by @s
camera back
wait 4
shot golem_fire_0
wait 4
