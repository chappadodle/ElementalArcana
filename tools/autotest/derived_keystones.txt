# The derived elements' keystones (see their spec), each on a grass platform:
# 1: Refraction (Crystal) at midnight: a skeleton 10 blocks off shoots at the mage (Resistance keeps
#    them up); every third arrow that reaches them comes back, so the skeleton's own health falls
#    (logged before and after 25 seconds, time for a few arrows to reach them);
# 2: Conductor (Lightning): a Thunderclap catches a husk 3 blocks off; a second husk 7 blocks off is
#    out of its reach but within 5 of the first, so only the arc hurts it (both logged);
# 3: Sunborn (Radiance): a Smite on a fresh husk at noon, another at midnight (the damage logged:
#    a quarter more at noon, a fifth less at midnight, about half again as much by day).
hud on
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd difficulty normal
cmd gamemode survival
cmd effect clear @s
cmd effect give @s minecraft:instant_health 1 10 true
cmd effect give @s minecraft:resistance 900 4 true
cmd effect give @s minecraft:saturation 900 0 true
cmd tp @s 160000.5 100 160000.5 -90 0
wait 60
cmd kill @e[type=!player,distance=..60]
cmd fill 159990 99 159985 160020 99 160015 minecraft:grass_block
cmd fill 159990 100 159985 160020 112 160015 minecraft:air
cmd tp @s 160000.5 100 160000.5 -90 0
# 1: Refraction.
cmd time set midnight
cmd arcana tree reset
cmd arcana affinity reset
cmd arcana awaken crystal
cmd arcana level set 40
cmd arcana tree take crystal_facet_1
cmd arcana tree take crystal_facet_2
cmd arcana tree take crystal_facet_3
cmd arcana tree take crystal_prismatic_ward
cmd arcana tree take crystal_refraction
cmd title @s clear
wait 20
cmd summon skeleton 160010.5 100 160000.5 {Tags:["eat"],PersistenceRequired:1b,HandItems:[{id:"minecraft:bow",count:1},{}]}
cmd arcana creaturelevel @e[tag=eat] set 40
wait 5
cmd execute as @e[tag=eat] run data get entity @s Health
wait 500
cmd execute as @e[tag=eat] run data get entity @s Health
camera back
wait 2
shot refraction
camera first
cmd kill @e[tag=eat]
cmd kill @e[type=arrow]
# 2: Conductor (back where it started: the arrows will have knocked the mage about).
cmd time set noon
cmd tp @s 160000.5 100 160000.5 -90 0
cmd arcana tree reset
cmd arcana affinity reset
cmd arcana awaken lightning
cmd arcana level set 40
cmd arcana tree take lightning_storm_1
cmd arcana tree take lightning_storm_2
cmd arcana tree take lightning_storm_3
cmd arcana tree take lightning_static_charge
cmd arcana tree take lightning_conductor
cmd arcana tree take lightning_spark_1
cmd arcana tree take lightning_spark_2
cmd arcana tree take thunderclap
cmd arcana cooldowns reset
cmd arcana mana fill
cmd title @s clear
wait 20
cmd summon husk 160003.5 100 160000.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
cmd summon husk 160007.5 100 160000.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
cmd arcana creaturelevel @e[tag=eat] set 40
wait 5
cmd execute as @e[tag=eat] run data get entity @s Health
select elementalarcana:thunderclap
press
release
wait 20
shot conductor
cmd execute as @e[tag=eat] run data get entity @s Health
cmd kill @e[tag=eat]
# 3: Sunborn.
cmd arcana tree reset
cmd arcana affinity reset
cmd arcana awaken radiance
cmd arcana level set 40
cmd arcana tree take radiance_dawn_1
cmd arcana tree take radiance_dawn_2
cmd arcana tree take radiance_dawn_3
cmd arcana tree take radiance_inner_light
cmd arcana tree take radiance_sunborn
cmd arcana cooldowns reset
cmd arcana mana fill
cmd title @s clear
cmd tp @s 160000.5 100 160000.5 -90 20
wait 20
cmd summon husk 160004.5 100 160000.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
cmd arcana creaturelevel @e[tag=eat] set 40
wait 5
cmd execute as @e[tag=eat] run data get entity @s Health
select elementalarcana:smite
press
release
wait 40
cmd execute as @e[tag=eat] run data get entity @s Health
cmd kill @e[tag=eat]
cmd time set midnight
cmd arcana cooldowns reset
cmd arcana mana fill
wait 20
cmd summon husk 160004.5 100 160000.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
cmd arcana creaturelevel @e[tag=eat] set 40
wait 5
cmd execute as @e[tag=eat] run data get entity @s Health
select elementalarcana:smite
press
release
wait 40
cmd execute as @e[tag=eat] run data get entity @s Health
cmd kill @e[tag=eat]
cmd arcana tree reset
cmd arcana affinity reset
cmd arcana awaken fire
cmd time set noon
cmd gamemode creative
cmd effect clear @s
