# The balance gauge (written by tools/gen_gauge.py: edit that, not this). Each fight: a build for
# the player's level, a foe at a set level 12 blocks ahead, the player standing and casting at it
# while /arcana gauge counts the damage both ways (the player is kept standing; the report says
# when they would have fallen).
hud on
cmd time set noon
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd gamerule doMobLoot false
cmd gamemode survival
cmd arcana affinity reset
cmd tp @s 0.5 100 0.5 0 0
wait 40

# a crypt's Revenant in a level-15 zone (story boss: 25; its element is random), against a player of 20
cmd kill @e[type=!player]
cmd kill @e[type=item]
cmd fill -21 99 -13 21 99 37 minecraft:stone
cmd fill -21 100 -13 21 112 37 minecraft:barrier
cmd fill -20 100 -12 20 112 36 minecraft:air
cmd tp @s 0.5 100 0.5 0 0
cmd effect clear @s
cmd arcana affinity reset
cmd arcana tree reset
cmd arcana stats reset
cmd arcana awaken fire
cmd arcana level set 20
cmd arcana stats spend 8 potency
cmd arcana stats spend 6 vitality
cmd arcana stats spend 4 ward
cmd arcana stats spend 1 focus
cmd arcana spell elementalarcana:fireball 5 cluster
cmd arcana tree take fire_burst_1
cmd arcana tree take fire_ember_1
cmd arcana tree take fire_spoke
cmd arcana tree take fire_sprite_1
cmd arcana tree take fire_burst_2
cmd arcana tree take fire_ember_2
cmd arcana tree take core_8
cmd arcana tree take fire_burst_3
cmd arcana tree take fire_ember_3
cmd arcana tree take core_7
cmd arcana tree take core_9
cmd arcana tree take fire_burst_4
cmd arcana tree take fire_kindled_soul
cmd arcana tree take core_6
cmd arcana tree take core_10
cmd item replace entity @s weapon.mainhand with elementalarcana:adept_staff[elementalarcana:element="fire"]
cmd item replace entity @s armor.head with elementalarcana:adept_hood
cmd item replace entity @s armor.chest with elementalarcana:adept_robe
cmd item replace entity @s armor.legs with elementalarcana:adept_trousers
cmd item replace entity @s armor.feet with elementalarcana:adept_boots
cmd arcana mana fill
cmd arcana cooldowns reset
cmd effect give @s minecraft:instant_health 1 10 true
cmd summon elementalarcana:revenant 0.5 100 12.5
cmd arcana creaturelevel @e[type=elementalarcana:revenant] set 15
cmd arcana creaturelevel @e[type=elementalarcana:revenant]
wait 20
select elementalarcana:fireball
cmd arcana gauge start
fight elementalarcana:revenant 150 20 10
wait 5
cmd arcana gauge stop
wait 10
