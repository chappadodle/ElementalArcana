# current code), each seen from the south and above.

cmd kill @e[type=!player]
cmd time set noon
cmd gamerule doDaylightCycle false
cmd weather clear
cmd gamemode spectator
cmd tp @s -30000 150 30000
wait 20
goto elementalarcana:wind_shrine 14 14
wait 200
shot shrine_wind
goto elementalarcana:fire_shrine 14 14
wait 200
shot shrine_fire
goto elementalarcana:earth_shrine 14 14
wait 200
shot shrine_earth
goto elementalarcana:water_shrine 14 14
wait 200
shot shrine_water
goto elementalarcana:ice_shrine 14 14
wait 200
shot shrine_ice
# Use one: stand on the platform facing the core (a spectator can't use blocks).

use

shot shrine_used
