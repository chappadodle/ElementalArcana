# Tremor: a shockwave runs east from the player, a block a tick, up a one-block step, past three
# husks in its way and one standing well off to the side; it's watched from the side. The husks'
# positions and health go to the log (the one off to the side keeps all 200). Then a second
# Tremor, as its caster sees it: first person and from behind. (The side view is from a glass
# block, so the camera stays put.)
hud on
cmd time set noon
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd gamerule doMobLoot false
cmd gamemode survival
cmd effect clear @s
cmd effect give @s minecraft:instant_health 1 10 true
cmd effect give @s minecraft:resistance 600 4 true
cmd tp @s 0.5 100 0.5 -90 10
wait 40
cmd kill @e[type=!player]
cmd kill @e[type=item]
cmd fill -8 99 -12 30 99 12 minecraft:grass_block
cmd fill -8 100 -12 30 108 12 minecraft:air
cmd fill 5 100 -6 30 100 6 minecraft:grass_block
cmd setblock 6 102 -10 minecraft:glass
cmd arcana tree reset
cmd arcana affinity reset
cmd arcana awaken earth
cmd arcana level set 20
cmd arcana spell elementalarcana:tremor 1
cmd arcana mana fill
cmd arcana cooldowns reset
cmd title @s clear
wait 20
cmd summon husk 3.5 100 0.5 {Tags:["eat"],PersistenceRequired:1b,Health:200f,attributes:[{id:"minecraft:generic.max_health",base:200d}]}
cmd summon husk 7.5 101 -0.5 {Tags:["eat"],PersistenceRequired:1b,Health:200f,attributes:[{id:"minecraft:generic.max_health",base:200d}]}
cmd summon husk 10.5 101 1.0 {Tags:["eat"],PersistenceRequired:1b,Health:200f,attributes:[{id:"minecraft:generic.max_health",base:200d}]}
cmd summon husk 7.5 101 4.5 {Tags:["eat"],PersistenceRequired:1b,Health:200f,attributes:[{id:"minecraft:generic.max_health",base:200d}]}
cmd effect give @e[type=husk] minecraft:slowness 60 255 true
cmd effect give @e[type=husk] minecraft:weakness 60 255 true
wait 10
cmd execute as @e[type=husk] run data get entity @s Pos
select elementalarcana:tremor
press
release
wait 3
cmd tp @s 6 103 -10 0 15
wait 2
shot tremor_running
wait 3
shot tremor_step
wait 3
shot tremor_end
wait 8
shot tremor_spikes
wait 30
shot tremor_after
cmd execute as @e[type=husk] run data get entity @s Pos
cmd execute as @e[type=husk] run data get entity @s Health
cmd kill @e[type=husk]
cmd arcana cooldowns reset
cmd arcana mana fill
cmd tp @s 0.5 100 0.5 -90 12
cmd summon husk 6.5 101 -0.5 {Tags:["eat"],PersistenceRequired:1b}
cmd summon husk 9.5 101 1.0 {Tags:["eat"],PersistenceRequired:1b}
cmd effect give @e[type=husk] minecraft:slowness 60 255 true
cmd effect give @e[type=husk] minecraft:weakness 60 255 true
wait 20
press
release
wait 6
shot tremor_caster_view
camera back
wait 5
shot tremor_from_behind
camera first
wait 30
cmd kill @e[type=husk]
cmd gamerule doMobLoot true
wait 10
quit
