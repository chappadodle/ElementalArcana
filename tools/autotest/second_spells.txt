# The derived elements' second spells in game: Thunderclap among husks (one soaked); Sanctuary under
# a hurt player with a husk inside (the player is mended, the husk seared); Prism Ward against a
# skeleton's arrows at night (they fly back at it: its health drops). Health goes to the log.
hud on
cmd time set noon
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd gamerule naturalRegeneration false
cmd gamemode survival
cmd effect clear @s
cmd effect give @s minecraft:instant_health 1 10 true
cmd tp @s 0.5 100 0.5 -90 5
wait 40
cmd kill @e[type=!player]
cmd kill @e[type=item]
cmd fill -4 99 -12 30 99 12 minecraft:grass_block
cmd fill -4 100 -12 30 108 12 minecraft:air
cmd arcana tree reset
cmd arcana affinity reset
cmd arcana awaken crystal
cmd arcana awaken lightning
cmd arcana awaken radiance
cmd arcana level set 40
cmd arcana tree take lightning_spark_1
cmd arcana tree take lightning_spark_2
cmd arcana tree take thunderclap
cmd arcana tree take radiance_halo_1
cmd arcana tree take radiance_halo_2
cmd arcana tree take sanctuary
cmd arcana tree take crystal_geode_1
cmd arcana tree take crystal_geode_2
cmd arcana tree take prism_ward
cmd arcana mana fill
cmd title @s clear
wait 40
cmd summon husk 3.5 100 0.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
cmd summon husk 2.5 100 3.5 {NoAI:1b,Tags:["eat","eawet"],PersistenceRequired:1b}
cmd summon husk 2.5 100 -3.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
cmd effect give @e[tag=eawet] elementalarcana:wet 30 0
wait 10
select elementalarcana:thunderclap
press
release
wait 2
shot thunderclap
wait 20
cmd execute as @e[type=husk] run data get entity @s Health
cmd kill @e[type=husk]
cmd damage @s 10 minecraft:generic
cmd summon husk 2.5 100 0.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
cmd arcana cooldowns reset
cmd arcana mana fill
wait 10
cmd data get entity @s Health
select elementalarcana:sanctuary
press
release
wait 50
shot sanctuary
cmd data get entity @s Health
cmd execute as @e[type=husk] run data get entity @s Health
cmd kill @e[type=husk]
cmd gamerule naturalRegeneration true
cmd effect give @s minecraft:instant_health 1 10 true
cmd effect give @s minecraft:resistance 600 4 true
cmd time set midnight
cmd summon skeleton 12.5 100 0.5 {Tags:["eat"],PersistenceRequired:1b}
cmd arcana cooldowns reset
cmd arcana mana fill
wait 10
select elementalarcana:prism_ward
press
release
wait 50
shot prism_ward
cmd execute as @e[type=skeleton] run data get entity @s Health
wait 60
cmd execute as @e[type=skeleton] run data get entity @s Health
cmd data get entity @s Health
cmd kill @e[type=skeleton]
cmd kill @e[type=arrow]
wait 10
quit
