# The kin elements' shrines: their chests' loot rolled and listed (their own Essence, their
# family's Catalyst now and then), then each located from the world spawn (the nearest Crystal,
# Lightning and Radiance shrine and how far; last, as the searches hold the server up).
gamerule doMobSpawning false
forceload add -16 -16 16 16
fill -12 99 -12 12 99 12 minecraft:stone
fill -12 100 -12 12 106 12 minecraft:air
kill @e[type=!player]
execute positioned 0 101 0 run loot spawn ~ ~ ~ loot elementalarcana:chests/shrine_crystal
execute positioned 0 101 0 run loot spawn ~ ~ ~ loot elementalarcana:chests/shrine_lightning
execute positioned 0 101 0 run loot spawn ~ ~ ~ loot elementalarcana:chests/shrine_radiance
sleep 1
execute as @e[type=item] run data get entity @s Item
kill @e[type=item]
forceload remove all
locate structure elementalarcana:crystal_shrine
locate structure elementalarcana:lightning_shrine
locate structure elementalarcana:radiance_shrine
