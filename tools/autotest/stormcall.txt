# Stormcall (see the spec) on a grass platform at noon: a mage of Lightning (level 40) calls the
# storm among three husks of its level (base health 100, more for their level: logged): one beside
# them, one 8 blocks off, one soaked 11 blocks off. Shots of the cloud and a strike; the husks'
# health logged three times over the 10 seconds (the soaked one struck first and hardest, then
# the bolts spread among all three; the one under the cloud gets soaked too); the cloud gone after.
hud on
cmd time set noon
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd gamemode survival
cmd effect clear @s
cmd effect give @s minecraft:instant_health 1 10 true
cmd effect give @s minecraft:resistance 600 4 true
cmd tp @s 142000.5 100 142000.5 -90 5
wait 60
cmd kill @e[type=!player,distance=..60]
cmd fill 141990 99 141985 142020 99 142015 minecraft:grass_block
cmd fill 141990 100 141985 142020 112 142015 minecraft:air
cmd tp @s 142000.5 100 142000.5 -90 5
cmd arcana tree reset
cmd arcana affinity reset
cmd arcana awaken lightning
cmd arcana level set 40
cmd arcana tree take lightning_cloud_1
cmd arcana tree take lightning_cloud_2
cmd arcana tree take stormcall
cmd arcana mana fill
cmd title @s clear
wait 40
cmd summon husk 142002.5 100 142000.5 {NoAI:1b,Tags:["eat","near"],PersistenceRequired:1b,Health:100f,attributes:[{id:"minecraft:generic.max_health",base:100d}]}
cmd summon husk 142008.5 100 142003.5 {NoAI:1b,Tags:["eat","far"],PersistenceRequired:1b,Health:100f,attributes:[{id:"minecraft:generic.max_health",base:100d}]}
cmd summon husk 142010.5 100 141996.5 {NoAI:1b,Tags:["eat","soaked"],PersistenceRequired:1b,Health:100f,attributes:[{id:"minecraft:generic.max_health",base:100d}]}
cmd arcana creaturelevel @e[tag=eat] set 40
cmd arcana creaturelevel @e[tag=eat]
cmd effect give @e[tag=soaked] elementalarcana:wet 30 0
cmd arcana cooldowns reset
cmd arcana mana fill
wait 10
select elementalarcana:stormcall
press
release
wait 12
look -90 -40
wait 3
shot stormcall_cloud
camera back
look -90 -18
wait 6
shot stormcall_cloud_back
look -90 5
wait 16
cmd execute as @e[tag=eat] run data get entity @s Health
wait 3
shot stormcall_strike
camera first
wait 60
cmd execute as @e[tag=eat] run data get entity @s Health
cmd execute as @e[tag=eat] run data get entity @s active_effects
wait 120
cmd execute as @e[tag=eat] run data get entity @s Health
look -90 -60
wait 5
shot stormcall_after
cmd kill @e[tag=eat]
cmd arcana tree reset
cmd arcana affinity reset
cmd arcana awaken fire
cmd gamemode creative
cmd effect clear @s
wait 10
quit
