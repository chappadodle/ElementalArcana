# The derived elements: their wisps summon (innate to their element), get a level, can't be
# attuned to another element and die cleanly. Creatures Attuned to them (only by command: nothing
# spawns Attuned to them) cast their spells at a villager 8 blocks away: Prism Bolt, Chain
# Lightning, Smite. The three pairs stand 20 blocks apart, out of Chain Lightning's reach.
time set midnight
gamerule doMobSpawning false
forceload add -16 -16 16 48
fill -12 99 -12 12 99 44 minecraft:stone
fill -12 100 -12 12 106 44 minecraft:air
summon elementalarcana:crystal_wisp 6 103 -8
summon elementalarcana:lightning_wisp 4 103 -8
summon elementalarcana:radiance_wisp 2 103 -8
sleep 2
arcana creaturelevel @e[type=elementalarcana:crystal_wisp]
arcana creaturelevel @e[type=elementalarcana:lightning_wisp]
arcana creaturelevel @e[type=elementalarcana:radiance_wisp]
# Refused: a wisp is innate to its element.
arcana attune @e[type=elementalarcana:lightning_wisp] wind adept
kill @e[type=elementalarcana:crystal_wisp]
kill @e[type=elementalarcana:lightning_wisp]
kill @e[type=elementalarcana:radiance_wisp]
sleep 1
summon villager 8.5 100 0.5 {NoAI:1b,Tags:["eatest","eacrystal"]}
summon zombie 0.5 100 0.5 {Tags:["eatest","eacaster1"],attributes:[{id:"minecraft:generic.movement_speed",base:0.0}]}
summon villager 8.5 100 20.5 {NoAI:1b,Tags:["eatest","ealightning"]}
summon zombie 0.5 100 20.5 {Tags:["eatest","eacaster2"],attributes:[{id:"minecraft:generic.movement_speed",base:0.0}]}
summon villager 8.5 100 40.5 {NoAI:1b,Tags:["eatest","earadiance"]}
summon zombie 0.5 100 40.5 {Tags:["eatest","eacaster3"],attributes:[{id:"minecraft:generic.movement_speed",base:0.0}]}
arcana attune @e[tag=eacaster1] crystal magus
arcana attune @e[tag=eacaster2] lightning magus
arcana attune @e[tag=eacaster3] radiance magus
sleep 10
data get entity @e[tag=eacrystal,limit=1] Health
data get entity @e[tag=ealightning,limit=1] Health
data get entity @e[tag=earadiance,limit=1] Health
kill @e[tag=eatest]
kill @e[type=elementalarcana:spell_projectile]
gamerule doMobSpawning true
forceload remove all
