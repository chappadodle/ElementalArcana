# Fire's moves: an Ember Sprite placed before a row of husks spits bolts at them; Pyronado's three
# wheels orbit the player among husks, burning and throwing them. Health goes to the log.
hud on
cmd time set midnight
cmd weather clear
cmd gamerule doDaylightCycle false
cmd gamerule doMobSpawning false
cmd gamerule doMobLoot false
cmd gamemode survival
cmd effect clear @s
cmd effect give @s minecraft:instant_health 1 10 true
cmd effect give @s minecraft:resistance 600 4 true
cmd tp @s 0.5 100 0.5 -90 10
wait 40
cmd kill @e[type=!player]
cmd kill @e[type=item]
cmd fill -8 99 -12 30 99 12 minecraft:grass_block
cmd fill -8 100 -12 30 108 12 minecraft:air
cmd arcana tree reset
cmd arcana affinity reset
cmd arcana awaken fire
cmd arcana level set 20
cmd arcana tree take fire_sprite_1
cmd arcana tree take ember_sprite
cmd arcana tree take fire_burst_1
cmd arcana tree take fire_burst_2
cmd arcana tree take fire_burst_3
cmd arcana tree take fire_burst_4
cmd arcana tree take flame_burst
cmd arcana mana fill
cmd arcana cooldowns reset
cmd title @s clear
wait 30
cmd summon husk 14.5 100 -2.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
cmd summon husk 14.5 100 2.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
cmd summon husk 16.5 100 0.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
wait 10
select elementalarcana:ember_sprite
press
release
wait 30
shot ember_sprite
wait 60
shot ember_sprite_bolts
cmd execute as @e[type=husk] run data get entity @s Health
cmd kill @e[type=husk]
cmd arcana level set 30
cmd arcana mana fill
cmd arcana cooldowns reset
cmd summon husk 3.5 100 2.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
cmd summon husk -2.5 100 2.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
cmd summon husk 0.5 100 -3.5 {NoAI:1b,Tags:["eat"],PersistenceRequired:1b}
wait 10
select elementalarcana:flame_burst
press
release
wait 12
camera back
wait 8
shot pyronado
camera first
wait 40
cmd execute as @e[type=husk] run data get entity @s Health
cmd kill @e[type=husk]
cmd gamerule doMobLoot true
wait 10
quit
